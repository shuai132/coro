cmake_minimum_required(VERSION 3.15)
project(coro_exception_configuration CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../warnings.cmake")

# Set the compiler state before adding the library, as a parent project or toolchain would.
if (MSVC)
    if (COMPILER_EXCEPTIONS)
        string(APPEND CMAKE_CXX_FLAGS " /EHsc")
    else ()
        string(APPEND CMAKE_CXX_FLAGS " /EHs-c-")
        add_compile_definitions(_HAS_EXCEPTIONS=0)
    endif ()
else ()
    if (COMPILER_EXCEPTIONS)
        string(APPEND CMAKE_CXX_FLAGS " -fexceptions")
    else ()
        string(APPEND CMAKE_CXX_FLAGS " -fno-exceptions")
    endif ()
endif ()

if (DEFINED PARENT_DISABLE_EXCEPTION)
    set(CORO_DISABLE_EXCEPTION "${PARENT_DISABLE_EXCEPTION}")
endif ()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../.." coro)
if (CORO_DISABLE_EXCEPTION)
    set(actual ON)
else ()
    set(actual OFF)
endif ()
if (NOT actual STREQUAL EXPECT_DISABLED)
    message(FATAL_ERROR "Expected CORO_DISABLE_EXCEPTION=${EXPECT_DISABLED}, found ${actual}")
endif ()

add_executable(exception_consumer main.cpp)
target_link_libraries(exception_consumer PRIVATE coro)
target_include_directories(exception_consumer PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../detail")
target_compile_definitions(exception_consumer PRIVATE EXPECT_DISABLED=$<BOOL:${EXPECT_DISABLED}>)
enable_testing()
add_test(NAME exception_consumer_run COMMAND exception_consumer)
